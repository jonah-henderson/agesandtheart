# Does a colony found itself? Two scarabs dropped over a site laid by hand in a colony-ready Age, the clock
# sprinted, and then where each of them lives.
# Run with: scripts/drive-server.sh scripts/checks/scarab-colony.txt
#
# The site is built on a raised block of ground, so it is the top of every column in it:
# mud in the middle, sand around it, torchflowers on grass beyond, all under a jungle `fillbiome` makes. A
# scarab should claim a mud column (it becomes `agesandtheart:scarab_nest`, mud to look at), stack
# mud on it out of the sand with one course of packed mud partway up, and graze the flowers. Read `home`, `satiety` and `carried` off each.

age write scarabcolony 7 age jungle biomes torchflowers in jungle mud pits in jungle sand pits in jungle
execute in agesandtheart:scarabcolony run forceload add -64 -64 64 64
execute in agesandtheart:scarabcolony run fillbiome -16 190 -16 16 215 16 minecraft:jungle
# Solid down to the Age's own ground, which is about y=150 here: a platform with an underside is a
# ceiling to anything that wanders beneath it, which no real ground has.
execute in agesandtheart:scarabcolony run fill -12 150 -12 12 199 12 minecraft:dirt
execute in agesandtheart:scarabcolony run fill -12 130 -12 12 149 12 minecraft:dirt replace #minecraft:air
execute in agesandtheart:scarabcolony run fill -12 200 -12 12 200 12 minecraft:grass_block
execute in agesandtheart:scarabcolony run fill -6 200 -6 6 200 6 minecraft:sand
execute in agesandtheart:scarabcolony run fill -2 200 -2 2 200 2 minecraft:mud
execute in agesandtheart:scarabcolony run fill -12 201 -12 -8 201 12 minecraft:torchflower
execute in agesandtheart:scarabcolony run time set noon
execute in agesandtheart:scarabcolony run summon agesandtheart:scarab 0 205 0
execute in agesandtheart:scarabcolony run summon agesandtheart:scarab 1 205 1
execute in agesandtheart:scarabcolony if entity @e[type=agesandtheart:scarab]

# A minute and a half of the Age's time.
tick sprint 1800
execute in agesandtheart:scarabcolony if entity @e[type=agesandtheart:scarab]
execute in agesandtheart:scarabcolony positioned 0 200 0 run data get entity @e[type=agesandtheart:scarab,sort=nearest,limit=1]
execute in agesandtheart:scarabcolony positioned 0 200 0 run data get entity @e[type=agesandtheart:scarab,sort=furthest,limit=1]

# Three minutes more, ending before dusk: the pillars should be rising, the sand going, and a young one or
# two bred onto the mud that is left.
tick sprint 4000
execute in agesandtheart:scarabcolony positioned 0 200 0 run data get entity @e[type=agesandtheart:scarab,sort=nearest,limit=1] home
execute in agesandtheart:scarabcolony positioned 0 200 0 run data get entity @e[type=agesandtheart:scarab,sort=furthest,limit=1] home
execute in agesandtheart:scarabcolony if entity @e[type=agesandtheart:scarab]

# Night: every housed scarab should go into its pillar beside the chamber, and the count fall to none. Then
# morning, when they come out where they went in.
execute in agesandtheart:scarabcolony run time set midnight
tick sprint 1200
execute in agesandtheart:scarabcolony if entity @e[type=agesandtheart:scarab]
execute in agesandtheart:scarabcolony run time set day
tick sprint 1200
execute in agesandtheart:scarabcolony if entity @e[type=agesandtheart:scarab]

# Counted by replacing, which is destructive and fine in a world about to be thrown away: how many nests,
# how much pillar, and how much of the sand is left.
execute in agesandtheart:scarabcolony run fill -2 200 -2 2 200 2 minecraft:dirt replace agesandtheart:scarab_nest
execute in agesandtheart:scarabcolony run fill -2 201 -2 2 208 2 minecraft:dirt replace minecraft:mud
execute in agesandtheart:scarabcolony run fill -2 201 -2 2 208 2 minecraft:dirt replace minecraft:packed_mud
execute in agesandtheart:scarabcolony run fill -6 200 -6 6 200 6 minecraft:dirt replace minecraft:sand
